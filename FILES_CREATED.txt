    --nep-potential nep_LiF_trained.txt \
    --test-cif LiF_simplecubic.cif

Use in VibroML:
  python vibroml/main.py \
    --cif LiF_simplecubic.cif \
    --engine gpumd \
    --checkpoint-path nep_LiF_trained.txt \
    --phonon
